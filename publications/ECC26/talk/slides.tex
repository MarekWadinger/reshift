% ===================== 2. THE MONITORING LOOP ============================
% PRESENTER: the setting — model-based monitoring, and the vocabulary for
% the whole talk. An identifier builds a model of the plant from inputs and
% measurements. The detector compares the plant against that model: the
% mismatch is the residual, windows of residuals form the detection
% statistic, the statistic crossing a threshold raises an alarm to the
% operator. Ideally the controller consumes the same model (MPC) — one model
% serving both. Hold this picture; the next slide is about what happens to
% that model over the years.

% ===================== 6b. CURRENT ONLINE SOLUTION FAILS =================

% ===================== 6c. OUR FIX: TRUNCATE ONLINE ======================
\begin{frame}{Our fix: truncate online — and keep the model aligned}
  % PRESENTER: truncation keeps only r directions, and those directions
  % drift as new data streams in. So each update we measure how much the
  % kept directions rotated (K_k), and apply that rotation to yesterday's model —
  % three cheap rotations and A-bar is valid in today's directions. That is
  % the entire contribution: truncation that survives streaming.
  \centering
  \resizebox{!}{0.72\textheight}{%
  \begin{tikzpicture}[scale=0.8, font=\sffamily,
    op/.style={font=\large, text=tgrey!80!black},
    cp/.style={font=\small, text=tgrey, align=left}]
    \def\P{1.0}\def\Q{0.6}

    % Row 1: split the rotation K by state (p) and control (q)
    \node[op] at (0.0,-0.8) {$K_k=$};
    \mblk{1.55}{0}{\P}{\P}{tblue}{$K_{pp}$}
    \mblk{2.55}{0}{\Q}{\P}{tgrey}{$K_{pq}$}
    \mblk{1.55}{-1.0}{\P}{\Q}{tgrey}{$K_{qp}$}
    \mblk{2.55}{-1.0}{\Q}{\Q}{tgreen}{$K_{qq}$}
    \lbr{1.45}{0.14}{-1.74} \rbr{3.25}{0.14}{-1.74}
    \node[op,anchor=west] at (3.5,-0.8) {$=\tilde U_k^\top\,\tilde U_{k-1}$\textsuperscript{\textdagger}};
    % PRESENTER: split by state (p) & control (q) subspaces, r = p + q;
    % only the diagonal blocks K_pp, K_qq are used.
    \node[cp,anchor=west,text width=5.0cm] at (7.0,-0.8)
      {how much the kept\\directions rotated this step};

    % Row 2: rotate the state operator (both sides by K_pp)
    \begin{scope}[yshift=-3.0cm]
      \node[op] at (0.0,-0.5) {$\tilde A_k\leftarrow$};
      \mblk{1.2}{0}{\P}{\P}{tblue}{$K_{pp}$}
      \node[op] at (2.35,-0.5) {$\cdot$};
      \mblk{2.6}{0}{\P}{\P}{tblue}{$\tilde A_k$}
      \node[op] at (3.75,-0.5) {$\cdot$};
      \mblk{4.0}{0}{\P}{\P}{tblue}{$K_{pp}^\top$}
      \node[op] at (5.15,-0.5) {$=$};
      \mblk{5.55}{0}{\P}{\P}{tblue}{$\tilde A_k$}
      \node[cp,anchor=west,text width=4.4cm] at (7.0,-0.5)
        {rotate the \textbf{state} operator};
    \end{scope}

    % Row 3: rotate the control operator (left K_pp, right K_qq)
    \begin{scope}[yshift=-5.0cm]
      \node[op] at (0.0,-0.5) {$\tilde B_k\leftarrow$};
      \mblk{1.2}{0}{\P}{\P}{tblue}{$K_{pp}$}
      \node[op] at (2.35,-0.5) {$\cdot$};
      \mblk{2.6}{0}{\Q}{\P}{tgreen}{$\tilde B_k$}
      \node[op] at (3.35,-0.5) {$\cdot$};
      \mblk{3.6}{0}{\Q}{\Q}{tgreen}{$K_{qq}$}
      \node[op] at (4.45,-0.5) {$=$};
      \mblk{4.85}{0}{\Q}{\P}{tgreen}{$\tilde B_k$}
      \node[cp,anchor=west,text width=4.6cm] at (7.0,-0.5)
        {rotate the \textbf{control} operator};
    \end{scope}

    % Row 4: assemble the augmented operator  A-bar = [ A | B ]
    \begin{scope}[yshift=-6.7cm]
      \node[op] at (0.0,-0.5) {$\tilde{\bar A}_k=$};

  \end{tikzpicture}}%

% ===================== 8. GUARANTEES =====================================
\begin{frame}{Two guarantees you can stand behind}
  % Figures from figs/fig_guarantees.py: left = measured update wall time under
  % the O(mr^2) envelope + flat over the stream; right = the frames/score_
  % synthetic — a sharp change peaks exactly c samples after onset (any c);
  % slower transitions (sharp <-> over 90) only push the peak later: delay >= c.
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{block}{Cheap: cost $\mathcal{O}(mr^2)$ per update}
        With $m$ sensors and a model of rank $r$ (a few dominant directions, $r \ll m$):
        a handful of operations per sample, \textbf{independent of history length}.\\
        Runs inside the sample interval.
      \end{block}
      \centering
      \includegraphics[width=0.92\linewidth,height=0.52\textheight,keepaspectratio]{fig_cost_mr2.pdf}
    \end{column}
    \begin{column}{0.5\linewidth}
      \begin{block}{Predictable: detection delay $\ge c$}
        The detection statistic \textbf{peaks about one \textcolor{tgreen}{test window} ($c$ samples)} after a persistent change.\\
        You choose $c$ — the test window is \textbf{your reaction budget}.
      \end{block}
